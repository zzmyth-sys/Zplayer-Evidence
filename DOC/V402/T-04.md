async fail(reason, code, message): Promise<boolean> {
  if (this.inFlight) { return false; }                 // InFlight 静默
  if (this.deps.lastItem() === null) { return false; }
  if (非 HLS) { publish Fatal; return false; }         // 非 HLS 不吞错
