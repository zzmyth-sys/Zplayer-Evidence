## 1. 目标 / 验收标准

| # | 目标 | 验收（可判定） |
|---|---|---|
| A1 | `prepareAsync` 返回不得视为播放成功 | 返回后状态仅 `Initialized`；未进入 `Playing` 前由停滞看门狗兜底 |
| A2 | HLS 切片 504/超时/断连走运行期降级 | 传输失败只进 `failHlsStream`，不置 `Error`、不发 Fatal |
| A3 | 降级可回退 SMB 代理 | 先销毁死实例，再以新 gen 取源 `avoid:['SFF']` |
| A4 | 非 HLS / 解码失败不吞错 | 仍走 Fatal 出口（重取源对解码失败无意义） |

### 2.6 运行期唯一降级入口（单飞 + 非 HLS 不吞错）— `PlayerAssembly.ets:215-256`

```ts
async failHlsStream(reason: string, code: number, message: string): Promise<boolean> {
  if (this.hlsFailInFlight) { return false; }             // 单飞：一个死流只降级一次
  const item: MediaItem | null = this.rt.lastItem;
  if (item === null) { return false; }
  if (this.rt.lastProtocol !== 'HLS' && !isHlsUrlPath(this.rt.lastUrl)) {
    this.publish({ kind: DomainEventKind.ErrorRaised,
      error: { code, level: ErrorLevel.Fatal, message } });
    return false;
  }
  this.hlsFailInFlight = true;
  try {
    const gen = this.intent.gen();
    const reqId = this.init.currentReqId();
    const ok = await this.downgrade.downgradeFromHls(item, gen, reqId, reason);
    if (ok) {                                             // 降级成功：补 scope，重挂看门狗
      const scope = new RequestScope(reqId, this.intent.gen(), item.uri);
      scope.bindEpoch(this.engine.getEpoch());
      this.playback.armStallWatchdogScope(scope, smbStreamHint());
    } else if (this.intent.isCurrent(gen, item.uri)) {    // 同请求且降级失败：可见 Fatal
      this.publish({ kind: DomainEventKind.ErrorRaised,
        error: { code, level: ErrorLevel.Fatal, message } });
    }
    return ok;
  } finally {
    this.hlsFailInFlight = false;
  }
}
```
### 2.7 控制器：先销毁死实例，再以新 gen 取源（`avoid:['SFF']`）— `SourceDowngradeCtl.ets:45-84`

```ts
downgradeFromHls(item, gen, reqId, reason): Promise<boolean> {
  return this.run(item, gen, reqId, reason, true);   // 运行期：disposeFirst=true
}

private async run(item, gen, reqId, reason, disposeFirst): Promise<boolean> {
  if (!this.deps.isCurrent(gen, item.uri)) { return false; }   // 已切歌，不降级
  const key = item.uri + '#' + gen.toString();
  if (this.tried.has(key)) { return false; }                    // 同 gen 只试一次
  this.tried.add(key);
  if (disposeFirst) { await this.releaseBackend(); }             // 释放失败不阻断
  const nextGen = this.deps.beginGen(item.uri);
  const outcome = await this.deps.prepare(item, nextGen,
    Deadline.after(OPEN_TOTAL_MS), reqId, ['SFF']);              // 新预算 + 避开 SFF
  if (outcome === 'Ok') { return true; }
  this.tried.delete(key);
  this.deps.publish({ kind: DomainEventKind.PlaybackError,
    error: { code: -4, level: ErrorLevel.Recoverable, message: 'hls downgrade failed' } });
  return false;
}
```
